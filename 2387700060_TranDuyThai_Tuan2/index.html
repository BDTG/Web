function updateInfo() {
    const name = prompt("Nhập tên mới:", "Trần Duy Thái");
    const mssv = prompt("Nhập MSSV mới:", "2387700060");
    const className = prompt("Nhập lớp mới:", "Kỹ thuật phần mềm");
    
    if (name) document.getElementById('display-name').innerText = name;

}
